similarity = FileAttachment("data/similarity.csv").csv({ typed: true })
albumsRaw = FileAttachment("data/albums.csv").csv({ typed: true })

albums = albumsRaw.filter(d => {
  const name = `${d.title} (${d.year})`
  return similarity.some(s => s["Album 1"] === name || s["Album 2"] === name)
})

viewof selectedArtist = Inputs.select([...new Set(albums.map(d => d.artist))], {
  label: "Select an artist:"
})

filteredAlbums = albums.filter(d => d.artist === selectedArtist)

viewof selected = Inputs.select(filteredAlbums, {
  label: "Select an album:",
  format: d => `${d.title} (${d.year})`
})

Cover Color Similarity

selectedAlbumName = selected ? `${selected.title} (${selected.year})` : ""

matches = similarity
  .filter(d => d["Album 1"] === selectedAlbumName || d["Album 2"] === selectedAlbumName)
  .map(d => {
    const isFirst = d["Album 1"] === selectedAlbumName;
    const albumName = isFirst ? d["Album 2"] : d["Album 1"];
    const artistName = isFirst ? d["Artist 2"] : d["Artist 1"];
    const info = albums.find(a => `${a.title} (${a.year})` === albumName);
    return {
      album: albumName,
      artist: artistName,
      similarity: d["Similarity"],
      cover: info ? info.cover_filename : null
    };
  })
  .sort((a, b) => b.similarity - a.similarity)

mostSimilar = matches[0]
mostDifferent = matches[matches.length - 1]

html`
${!mostSimilar ? html`<p style="color: #888; font-style: italic;">No similarity data available for this album.</p>` : html`
<div style="display: flex; gap: 24px; flex-wrap: wrap; margin-top: 20px;">
  <div class="album-card" style="flex: 1; min-width: 240px; max-width: 300px;">
    <div class="similarity-badge badge-similar">★ Most Similar • ${mostSimilar.similarity}%</div>
    <img src="data/${mostSimilar.cover}" alt="${mostSimilar.album}" style="width: 100%; border-radius: 8px; box-shadow: 0 2px 8px rgba(0,0,0,0.1);">
    <p style="margin-top: 12px; font-weight: 600; margin-bottom: 2px;">${mostSimilar.album}</p>
    <p style="color: #6c757d; font-size: 0.9rem; margin-top: 0;">${mostSimilar.artist}</p>

    <p style="color: #6c757d; font-size: 0.9rem; margin-top: 0;">${mostDifferent.artist}</p>
  </div>
</div>
`}
