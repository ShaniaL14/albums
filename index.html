albums = FileAttachment("data/albums.csv").csv({ typed: true })

viewof selectedArtist = Inputs.select([...new Set(albums.map(d => d.artist))], {
  label: "Select an artist:"
})

filteredAlbums = albums.filter(d => d.artist === selectedArtist)

viewof selected = Inputs.select(filteredAlbums, {
  label: "Select an album:",
  format: d => `${d.title} (${d.year})`
})

html`<div style="margin-top: 20px; margin-bottom: 30px;">
  <img src="data/${selected?.cover_filename}" alt="${selected?.title}" style="max-width: 400px; width: 100%; border-radius: 8px; box-shadow: 0 4px 6px rgba(0,0,0,0.1);" onerror="this.style.display='none'; this.nextElementSibling.style.display='block';">
  <p style="display:none; color: #888; font-style: italic;">Cover image not available.</p>
</div>`

Cover Color Similarity

similarity = FileAttachment("data/similarity.csv").csv({ typed: true })

selectedAlbumName = selected ? `${selected.title} (${selected.year})` : ""

matches = similarity
  .filter(d => d["Album 1"] === selectedAlbumName || d["Album 2"] === selectedAlbumName)
  .map(d => {
    const isFirst = d["Album 1"] === selectedAlbumName;
    const albumName = isFirst ? d["Album 2"] : d["Album 1"];
    const artistName = isFirst ? d["Artist 2"] : d["Artist 1"];
    const info = albums.find(a => `${a.title} (${a.year})` === albumName);
    return {
      album: albumName,
      artist: artistName,
      similarity: d["Similarity"],
      cover: info ? info.cover_filename : null
    };
  })
  .sort((a, b) => b.similarity - a.similarity)

mostSimilar = matches[0]
mostDifferent = matches[matches.length - 1]

html`
${!mostSimilar ? html`<p style="color: #888; font-style: italic;">No similarity data available for this album.</p>` : html`
<div style="display: flex; gap: 30px; flex-wrap: wrap; margin-top: 15px;">
  <div style="flex: 1; min-width: 220px; max-width: 300px; text-align: center;">
    <h4 style="color: #2e7d32;">Most Similar (${mostSimilar.similarity}%)</h4>
    <img src="data/${mostSimilar.cover}" alt="${mostSimilar.album}" style="width: 100%; border-radius: 8px; box-shadow: 0 4px 6px rgba(0,0,0,0.1);">
    <p style="margin-top: 8px; font-weight: bold; margin-bottom: 2px;">${mostSimilar.album}</p>
    <p style="color: #666; margin-top: 0;">${mostSimilar.artist}</p>
  </div>
  <div style="flex: 1; min-width: 220px; max-width: 300px; text-align: center;">
    <h4 style="color: #c62828;">Most Different (${mostDifferent.similarity}%)</h4>
    <img src="data/${mostDifferent.cover}" alt="${mostDifferent.album}" style="width: 100%; border-radius: 8px; box-shadow: 0 4px 6px rgba(0,0,0,0.1);">
    <p style="margin-top: 8px; font-weight: bold; margin-bottom: 2px;">${mostDifferent.album}</p>
    <p style="color: #666; margin-top: 0;">${mostDifferent.artist}</p>
  </div>
</div>
`}
`
